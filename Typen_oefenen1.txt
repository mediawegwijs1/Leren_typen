  const AMOUNTS = { time:[15,30,60,120], words:[10,25,50,100] };
  const MAX_EXTRA = 8;
  const $ = id => document.getElementById(id);

  const panel=$('speedPanel'), wrap=$('spWordsWrap'), wordsEl=$('spWords'), input=$('spInput');
  const testEl=$('spTest'), resultEl=$('spResult');
  const liveCount=$('spLiveCount'), liveWpm=$('spLiveWpm'), ghostTag=$('spGhostTag');
  const caret=document.createElement('div'); caret.id='spCaret';
  const ghostCaret=document.createElement('div'); ghostCaret.id='spGhost';

  let settings = { level:'standaard', mode:'time', amount:30 };
  try { Object.assign(settings, JSON.parse(localStorage.getItem('speedTypeSettings')||'{}')); } catch(e){}
  if(settings.level!=='moeilijk') settings.level='standaard';
  if(!AMOUNTS[settings.mode]) settings.mode='time';
  if(!AMOUNTS[settings.mode].includes(settings.amount)) settings.amount=AMOUNTS[settings.mode][1];

  let isOpen=false;
  // seps[i] = wat er na woord i getypt moet worden: ' ' (spatie), '\n' (Enter) of '' (einde tekst)
  let words=[], seps=[], typed=[], typedSeps=[], cur=0, wordEls=[];
  let shownWpm=null;
  let running=false, finished=false, startT=0, tick=null, raf=null;
  let correctKeys=0, wrongKeys=0, secKeys=0, secErr=0, lastSec=0;
  let history=[], events=[];
  let ghost=null, ghostIdx=0, lastRun=null;
  let gen={ left:0, last:'' };
  let lineH=0;

  // ---------- Tekst maken ----------
  function cap(w){ return w.charAt(0).toUpperCase()+w.slice(1); }
  function pick(arr){ return arr[Math.floor(Math.random()*arr.length)]; }
  // Moeilijk: zinnen met hoofdletter aan het begin, leestekens en Enter aan het eind van elke zin
  function genWords(n, isEnd){
    const out={ words:[], seps:[] };
    const hard = settings.level==='moeilijk';
    for(let i=0;i<n;i++){
      let w;
      do { w=pick(WORD_LIST); } while(w===gen.last);
      gen.last=w;
      let sep=' ';
      if(hard){
        const first = gen.left===0;
        if(first){ gen.left=4+Math.floor(Math.random()*6); w=cap(w); }
        gen.left--;
        const endOfText = isEnd && i===n-1;
        if(gen.left===0 || endOfText){
          w += pick(['.','.','.','?','!']);
          sep='\n';
          gen.left=0;
        }else if(!first && gen.left>=2){
          const r=Math.random();
          if(r<0.14) w+=',';
          else if(r<0.17) w+=':';
          else if(r<0.20) w='('+w+')';
          else if(r<0.22) w+='/'+pick(WORD_LIST);
        }
      }
      out.words.push(w); out.seps.push(sep);
    }
    if(isEnd && out.seps.length) out.seps[out.seps.length-1]='';
    return out;
  }

  // ---------- Weergave ----------
  function wordOk(i){ return typed[i]===words[i] && typedSeps[i]===seps[i]; }
  function renderWord(i){
    const el=wordEls[i]; if(!el) return;
    const w=words[i], t=typed[i]||'';
    el.className='spWord'+(i<cur && !wordOk(i) ? ' err':'');
    el.textContent='';
    const len=Math.max(w.length,t.length);
    for(let j=0;j<len;j++){
        const s=document.createElement('span');
        s.className='spL';
        if(j<w.length){
          s.textContent=w[j];
          if(j<t.length) s.classList.add(t[j]===w[j]?'ok':'bad');
        }else{
          s.textContent=t[j];
          s.classList.add('extra');
        }
        el.appendChild(s);
    }
    if(seps[i]==='\n'){
      const s=document.createElement('span');
      s.className='spL enter';
      s.textContent='↵';
      if(i<cur) s.classList.add(typedSeps[i]==='\n'?'ok':'bad');
      el.appendChild(s);
    }
  }
  function addWordEls(from){
    const frag=document.createDocumentFragment();
    for(let i=from;i<words.length;i++){
        const el=document.createElement('div');
        wordEls[i]=el; frag.appendChild(el);
        if(seps[i]==='\n'){
          const br=document.createElement('div');
          br.className='spBreak';
          frag.appendChild(br);
        }
    }
    wordsEl.insertBefore(frag, caret);
    for(let i=from;i<words.length;i++) renderWord(i);
  }
  function renderAll(){
    wordsEl.textContent='';
    wordsEl.append(caret, ghostCaret);
    wordEls=[];
    addWordEls(0);
    measure();
  }
  function measure(){
    if(wordEls[0]) lineH=wordEls[0].offsetHeight || lineH;
    positionAll();
  }
  function caretPos(w,c){
    const el=wordEls[w]; if(!el) return null;
    const L=el.children;
    if(!L.length) return {x:el.offsetLeft, y:el.offsetTop};
    if(c<L.length) return {x:L[c].offsetLeft, y:el.offsetTop};
    const last=L[L.length-1];
    return {x:last.offsetLeft+last.offsetWidth, y:el.offsetTop};
  }
  function positionAll(){
    const p=caretPos(cur,(typed[cur]||'').length);
    if(p){
        caret.style.left=p.x+'px'; caret.style.top=p.y+'px';
        if(lineH){
          const line=Math.round(p.y/lineH);
          wordsEl.style.transform=`translateY(${-Math.max(0,line-1)*lineH}px)`;
          input.style.top=Math.min(p.y, lineH)+'px';
        }
    }
    positionGhost();
  }
  function positionGhost(){
    if(!ghost){ ghostCaret.style.display='none'; return; }
    const ev=ghost.events[ghostIdx] || {w:0,c:0};
    const p=caretPos(ev.w, ev.c);
    if(!p){ ghostCaret.style.display='none'; return; }
    ghostCaret.style.display='block';
    ghostCaret.style.left=p.x+'px'; ghostCaret.style.top=p.y+'px';
  }

  // ---------- Statistiek ----------
  function correctChars(){
    let n=0;
    for(let i=0;i<cur;i++) if(wordOk(i)) n+=words[i].length+1;
    const t=typed[cur]||'', w=words[cur]||'';
    if(t && w.startsWith(t)) n+=t.length;
    return n;
  }
  function typedChars(){
    let n=cur;
    for(let i=0;i<=cur;i++) n+=(typed[i]||'').length;
    return n;
  }
  function elapsedSec(){ return running ? (performance.now()-startT)/1000 : 0; }
  function wpmAt(sec, chars){ return sec>0 ? chars/5/(sec/60) : 0; }

  function updateLive(){
    if(settings.mode==='time'){
        const left = running ? Math.max(0, Math.ceil(settings.amount-elapsedSec())) : settings.amount;
        liveCount.textContent=left;
    }else{
        liveCount.textContent=`${Math.min(cur,words.length)}/${words.length}`;
    }
    liveCount.className = running ? '' : 'muted';
    if(!running) liveWpm.textContent='';
  }
  // Live wpm rustig houden: alleen 1x per seconde bijwerken, vanaf 2 s, en zacht uitmiddelen
  function updateLiveWpm(sec){
    if(sec<2) return;
    const now=wpmAt(sec,correctChars());
    shownWpm = shownWpm===null ? now : shownWpm*0.7 + now*0.3;
    liveWpm.textContent=Math.round(shownWpm)+' wpm';
  }

  // ---------- Start / stop ----------
  function resetRun(text){
    clearInterval(tick); cancelAnimationFrame(raf);
    running=false; finished=false;
    words=text.words; seps=text.seps; typed=['']; typedSeps=[]; cur=0;
    correctKeys=0; wrongKeys=0; secKeys=0; secErr=0; lastSec=0; shownWpm=null;
    history=[]; events=[]; ghostIdx=0;
    input.value='';
    caret.classList.add('idle');
    ghostTag.style.display = ghost ? 'inline' : 'none';
    $('spTypeTip').innerHTML = settings.level==='moeilijk'
      ? 'Begin gewoon met typen · <kbd>spatie</kbd> = volgend woord · <kbd>Enter</kbd> bij ↵ = nieuwe zin'
      : 'Begin gewoon met typen · <kbd>spatie</kbd> = volgend woord';
    resultEl.hidden=true; testEl.hidden=false;
    renderAll();
    updateLive();
    focusInput();
  }
  function newText(){
    ghost=null;
    gen={left:0,last:''};
    const n = settings.mode==='words' ? settings.amount : 80;
    resetRun(genWords(n, settings.mode==='words'));
  }
  function retry(){
    if(!lastRun) return newText();
    ghost=lastRun;
    resetRun({ words:lastRun.words.slice(), seps:lastRun.seps.slice() });
  }
  function start(){
    running=true;
    startT=performance.now();
    caret.classList.remove('idle');
    tick=setInterval(onTick,50);
    raf=requestAnimationFrame(ghostLoop);
  }
  function pushSecond(s){
    history.push({ s, wpm:wpmAt(s,correctChars()), raw:secKeys*12, err:secErr });
    secKeys=0; secErr=0;
  }
  function onTick(){
    if(!running) return;
    const el=elapsedSec();
    const limit = settings.mode==='time' ? settings.amount : Infinity;
    while(lastSec < Math.min(Math.floor(el), limit)){ lastSec++; pushSecond(lastSec); updateLiveWpm(lastSec); }
    updateLive();
    if(settings.mode==='time' && el>=settings.amount) finish();
  }
  function ghostLoop(){
    if(!running || !ghost) return;
    const t=performance.now()-startT;
    const evs=ghost.events;
    let moved=false;
    while(ghostIdx+1<evs.length && evs[ghostIdx+1].t<=t){ ghostIdx++; moved=true; }
    if(moved) positionGhost();
    raf=requestAnimationFrame(ghostLoop);
  }

  function finish(){
    if(finished) return;
    const el = settings.mode==='time' ? settings.amount : elapsedSec();
    if(settings.mode==='words'){
        while(lastSec < Math.floor(el)){ lastSec++; pushSecond(lastSec); }
        if(el-lastSec>0.25 || history.length===0){
          const part=Math.max(el-lastSec, 0.001);
          history.push({ s:+el.toFixed(2), wpm:wpmAt(el,correctChars()), raw:secKeys*12/part, err:secErr });
        }
    }
    finished=true; running=false;
    clearInterval(tick); cancelAnimationFrame(raf);
    input.blur();

    let ok=0, bad=0, extra=0, missed=0;
    const lastIdx=Math.min(cur, words.length-1);
    for(let i=0;i<=lastIdx;i++){
        const w=words[i], t=typed[i]||'';
        for(let j=0;j<Math.max(w.length,t.length);j++){
          if(j<t.length){
            if(j>=w.length) extra++;
            else if(t[j]===w[j]) ok++;
            else bad++;
          }else if(i<cur) missed++;
        }
    }
    const totalKeys=correctKeys+wrongKeys;
    const run={
        words:words.slice(), seps:seps.slice(), events, history,
        wpm:Math.round(wpmAt(el,correctChars())),
        raw:Math.round(wpmAt(el,typedChars())),
        acc: totalKeys ? Math.round(correctKeys/totalKeys*100) : 0,
        time:el, chars:[ok,bad,extra,missed]
    };
    showResult(run, ghost);
    lastRun=run;
    if(typeof shootConfetti==='function' && (!ghost || run.wpm>ghost.wpm)) shootConfetti();
  }

  // ---------- Invoer ----------
  function recordEvent(){
    if(running) events.push({ t:performance.now()-startT, w:cur, c:(typed[cur]||'').length });
  }
  function setCurrent(seg){
    const w=words[cur], old=typed[cur]||'';
    seg=seg.slice(0, w.length+MAX_EXTRA);
    if(seg.length>old.length){
        let k=0; while(k<old.length && old[k]===seg[k]) k++;
        for(let i=k;i<seg.length;i++){
          secKeys++;
          if(i<w.length && seg[i]===w[i]) correctKeys++;
          else { wrongKeys++; secErr++; }
        }
    }
    typed[cur]=seg;
    renderWord(cur);
    if(settings.mode==='words' && cur===words.length-1 && seg===w){ recordEvent(); finish(); }
  }
  function commitWord(sep){
    secKeys++;
    typedSeps[cur]=sep;
    if(wordOk(cur) || (seps[cur]==='' && typed[cur]===words[cur])) correctKeys++; else { wrongKeys++; secErr++; }
    cur++;
    typed[cur]='';
    renderWord(cur-1);
    if(cur>=words.length){
        if(settings.mode==='words'){ cur=words.length-1; recordEvent(); finish(); return; }
    }
    if(settings.mode==='time' && words.length-cur<40){
        const from=words.length;
        const more=genWords(60,false);
        words.push(...more.words); seps.push(...more.seps);
        addWordEls(from);
    }
  }
  input.addEventListener('input',()=>{
    if(finished){ input.value=''; return; }
    const v=input.value.replace(/[\u00A0\n\r\t]/g,' ');
    if(!running && v.trim().length>0) start();
    if(!running){ input.value=''; return; }
    const parts=v.split(' ');
    for(let p=0;p<parts.length;p++){
        setCurrent(parts[p]);
        if(finished) return;
        if(p<parts.length-1 && typed[cur].length>0){
          commitWord(' ');
          if(finished) return;
        }
    }
    if(input.value!==typed[cur]) input.value=typed[cur];
    recordEvent();
    positionAll();
    ensureVisible();
    updateLive();
  });
  input.addEventListener('keydown',e=>{
    if(e.key==='Enter'){
        e.preventDefault();
        if(!running || finished || !(typed[cur]||'').length) return;
        commitWord('\n');
        if(finished) return;
        input.value='';
        recordEvent(); positionAll(); updateLive(); ensureVisible();
        return;
    }
    if(e.key==='Backspace' && input.value==='' && cur>0 && !finished && !wordOk(cur-1)){
        // terug naar het vorige woord als daar een fout in zit
        e.preventDefault();
        typed[cur]=''; renderWord(cur);
        cur--; renderWord(cur);
        input.value=typed[cur];
        recordEvent(); positionAll(); updateLive(); ensureVisible();
    }
  });
  input.addEventListener('paste',e=>{ e.preventDefault(); });
  input.addEventListener('blur',()=>{ if(isOpen && !finished) wrap.classList.add('blurred'); });
  input.addEventListener('focus',()=> wrap.classList.remove('blurred'));
  wrap.addEventListener('click',focusInput);

  function focusInput(){
    if(!isOpen || finished) return;
    input.focus({preventScroll:true});
    wrap.classList.remove('blurred');
  }
  // Houd de regel waarop je typt in beeld (ook met tablet-toetsenbord open of na zelf scrollen)
  function ensureVisible(){
    if(!isOpen || testEl.hidden) return;
    const vv=window.visualViewport;
    const visTop = vv ? vv.offsetTop : 0;
    const visH = vv ? vv.height : window.innerHeight;
    const r=wrap.getBoundingClientRect();
    const c=caret.getBoundingClientRect();
    const margin=Math.min(60, visH*0.1);
    if(r.top >= visTop+margin && c.bottom <= visTop+visH-margin) return;
    // Tekstvak (met de teller erboven) in het bovenste deel van het zichtbare scherm zetten
    const want = visTop + Math.max(margin, Math.min(visH*0.25, (visH-r.height)/2));
    window.scrollBy({ top: r.top - want, behavior:'auto' });
  }
  if(window.visualViewport){
    window.visualViewport.addEventListener('resize',()=>{
      if(isOpen && document.activeElement===input) setTimeout(ensureVisible,50);
    });
  }

  document.addEventListener('keydown',e=>{
    if(!isOpen) return;
    // Na de ronde start niets vanzelf opnieuw: alleen klikken op Opnieuw / Volgende tekst
    if(finished){
        if((e.key===' ' || e.key==='Enter') && panel.contains(document.activeElement)) e.preventDefault();
        return;
    }
    if(!finished && document.activeElement!==input && e.key.length===1 && !e.ctrlKey && !e.metaKey && !e.altKey){
        focusInput();
    }
  });
  window.addEventListener('resize',()=>{ if(isOpen){ measure(); if(finished && lastRun) drawChart(lastRun, chartGhost); } });

  // ---------- Instellingen ----------
  const amountsEl=$('spAmounts');
  function renderConfig(){
    panel.querySelectorAll('[data-level]').forEach(b=> b.classList.toggle('on', b.dataset.level===settings.level));
    panel.querySelectorAll('[data-mode]').forEach(b=> b.classList.toggle('on', b.dataset.mode===settings.mode));
    amountsEl.textContent='';
    AMOUNTS[settings.mode].forEach(a=>{
        const b=document.createElement('button');
        b.className='spOpt'+(a===settings.amount?' on':'');
        b.textContent=a;
        b.addEventListener('click',()=>{ settings.amount=a; changed(); });
        amountsEl.appendChild(b);
    });
  }
  function changed(){
    try{ localStorage.setItem('speedTypeSettings', JSON.stringify(settings)); }catch(e){}
    lastRun=null;
    renderConfig();

  function drawChart(run, prev){
    const cv=$('spChart');
    const W=cv.clientWidth, H=cv.clientHeight;
    if(!W || !H) return;
    const dpr=window.devicePixelRatio||1;
    cv.width=W*dpr; cv.height=H*dpr;
    const g=cv.getContext('2d');
    g.setTransform(dpr,0,0,dpr,0,0);
    g.clearRect(0,0,W,H);
    const data=run.history, gdata=prev ? prev.history : [];
    if(!data.length) return;
    const pad={l:52,r:40,t:12,b:30};
    const maxS=Math.max(data[data.length-1].s, gdata.length ? gdata[gdata.length-1].s : 0, 1);
    const minS=1;
    // rauwe snelheid per seconde uitmiddelen over 3 seconden zodat de lijn rustiger is
    const rawSmooth=data.map((d,i)=>{
      const win=data.slice(Math.max(0,i-1), i+2);
      return win.reduce((a,b)=>a+b.raw,0)/win.length;
    });
    let maxY=Math.max(10, ...data.map((d,i)=>Math.max(d.wpm,rawSmooth[i])), ...gdata.map(d=>d.wpm));
    maxY=Math.ceil(maxY*1.1/20)*20;
    const maxE=Math.max(1, ...data.map(d=>d.err));
    const X=s=> pad.l + (maxS===minS ? 0.5 : (s-minS)/(maxS-minS)) * (W-pad.l-pad.r);
    const Y=v=> pad.t + (1-v/maxY)*(H-pad.t-pad.b);
    const YE=v=> pad.t + (1-v/maxE)*(H-pad.t-pad.b);
    const mono='12px ui-monospace,Menlo,Consolas,monospace';

    g.font=mono; g.fillStyle='#94a3b8'; g.strokeStyle='#e2e8f0'; g.lineWidth=1;
    for(let i=0;i<=4;i++){
        const v=maxY*i/4, y=Y(v);
        g.beginPath(); g.moveTo(pad.l,y); g.lineTo(W-pad.r,y); g.stroke();
        g.textAlign='right'; g.textBaseline='middle'; g.fillText(Math.round(v), pad.l-8, y);
    }
    for(let i=0;i<=Math.min(maxE,4);i++){
        const v=Math.round(maxE*i/Math.min(maxE,4));
        g.textAlign='left'; g.fillText(v, W-pad.r+8, YE(v));
    }
    const span=Math.max(1, Math.ceil((maxS-minS+1)/15));
    g.textAlign='center'; g.textBaseline='top';
    for(let s=minS;s<=Math.floor(maxS);s+=span) g.fillText(s, X(s), H-pad.b+8);
    g.save(); g.translate(14,(H-pad.b+pad.t)/2); g.rotate(-Math.PI/2);
    g.textBaseline='middle'; g.fillText('woorden per minuut',0,0); g.restore();
    g.save(); g.translate(W-8,(H-pad.b+pad.t)/2); g.rotate(Math.PI/2);
    g.textBaseline='middle'; g.fillText('fouten',0,0); g.restore();

    function line(pts,color,width,dash,dots){
        if(!pts.length) return;
        g.save(); g.strokeStyle=color; g.lineWidth=width; g.setLineDash(dash||[]);
        g.lineJoin='round'; g.lineCap='round';
        g.beginPath(); g.moveTo(pts[0].x,pts[0].y);
        for(let i=1;i<pts.length;i++){
          const mx=(pts[i-1].x+pts[i].x)/2, my=(pts[i-1].y+pts[i].y)/2;
          g.quadraticCurveTo(pts[i-1].x,pts[i-1].y,mx,my);
        }
        g.lineTo(pts[pts.length-1].x,pts[pts.length-1].y);
        g.stroke();
        if(dots){ g.fillStyle=color; pts.forEach(p=>{ g.beginPath(); g.arc(p.x,p.y,2.5,0,Math.PI*2); g.fill(); }); }
        g.restore();
    }
    line(data.map((d,i)=>({x:X(d.s),y:Y(Math.min(rawSmooth[i],maxY))})), '#94a3b8', 2);
    if(gdata.length) line(gdata.map(d=>({x:X(d.s),y:Y(d.wpm)})), '#8b5cf6', 2, [6,5]);
    line(data.map(d=>({x:X(d.s),y:Y(d.wpm)})), '#f59e0b', 3, null, true);
    g.fillStyle='#ef4444'; g.font='bold 13px sans-serif'; g.textAlign='center'; g.textBaseline='middle';
    data.forEach(d=>{ if(d.err>0) g.fillText('✕', X(d.s), YE(d.err)); });
  }

    document.body.classList.remove('speed-mode');

  }

  return { open, close };
})();

function updateLineChecks(typedNorm){
  if(targetLines.length===0){ linesOK.textContent="0/0"; return; }
  const typedLines=typedNorm.split("\n");
  let ok=0;
  for(let i=0;i<targetLines.length;i++){
    const good=typedLines[i]===targetLines[i];
    if(good){
      lineChipsEls[i].classList.add('ok');
      lineChipsEls[i].classList.remove('pending');
      ok++;
    }else{
      lineChipsEls[i].classList.add('pending');
      lineChipsEls[i].classList.remove('ok');
    }
  }
  linesOK.textContent=`${ok}/${targetLines.length}`;

function compareAndRender(){
  let typed=normalizeInput(inputEl.value);
  if(currentLevel==='easy'){
    typed=typed.replace(/\n/g," ").replace(/\s+/g," ");
  }
  totalTyped=typed.length;

  const compTarget=(currentLevel==='easy') ? EASY_WORDS : target;
  let correct=0, wrong=0;
  const nextIndex=typed.length;

  const flatSpans=Array.from(typedLine.querySelectorAll('.glyph'));
  for(let i=0;i<flatSpans.length;i++){
    const span=flatSpans[i];
    const expected=compTarget[i];
    const got=typed[i];
    span.classList.remove('correct','wrong','next');
    if(i<typed.length){
      if(got===expected){ span.classList.add('correct'); correct++; }
      else{ span.classList.add('wrong'); wrong++; }
    }else if(i===nextIndex){
      span.classList.add('next');
    }
  }

  const acc= totalTyped>0 ? Math.round((correct/totalTyped)*100) : 0;
  currentAcc = acc;
  accEl.textContent=acc+"%";
  errEl.textContent=wrong;

  const nextCh=compTarget[nextIndex] || null;
  highlightNext(nextCh);

  if(currentLevel!=='easy'){ updateLineChecks(typed); }

  /* --- FIX V5.1: Laat de onzichtbare textarea de actieve letter volgen --- */
  const nextGlyph = typedLine.querySelector('.glyph.next');
  if (nextGlyph) {
    // 1. We plakken het onzichtbare tekstvak exact op de huidige letter.
    // De tablet (en het ingebouwde toetsenbord) zal nu uit ZICHZELF 
    // altijd zorgen dat deze specifieke hoogte goed in beeld blijft!
    inputEl.style.top = nextGlyph.offsetTop + 'px';
    
    // 2. Extra vangnet: Zodra we een behoorlijk stuk naar beneden gaan, 
    // sturen we toch nog even een vloeiend scroll-signaal.
    if (Math.abs(nextGlyph.offsetTop - lastGlyphTop) > 10) {
      nextGlyph.scrollIntoView({ behavior: 'smooth', block: 'center' });
      lastGlyphTop = nextGlyph.offsetTop;
    }
  }
  /* ------------------------------------------------------------------------ */

  if(typed===compTarget){
    const secs=(Date.now()-startTime)/1000;
    
    // NIEUW: Als we in Level 3 zitten en het typen is klaar, start dan de Kleur-fase
    if(currentLevel === 'style' && stylePhase === 'typing') {
      stylePhase = 'coloring';
      shootConfetti(); 
      startStylePhase();
      return; 
    }

    if(['easy','normal','hard'].includes(currentLevel)){
      levelStats[currentLevel]={done:true,time:secs,acc:acc,errors:wrong};
    }
    clearInterval(timer); started=false; inputEl.disabled=true;
    if(currentLevel==='easy'){
      btnNormal.disabled=false; btnNormal.classList.remove('disabled');
    }
    if(currentLevel==='normal'){
      btnHard.disabled=false; btnHard.classList.remove('disabled');
    }
    if(currentLevel==='hard'){

// -------- Confetti --------
const confettiCanvas=document.getElementById('confetti');
const ctx=confettiCanvas.getContext('2d');
let confettiPieces=[];

function resizeCanvas(){
  confettiCanvas.width=innerWidth;
  confettiCanvas.height=innerHeight;
}
window.addEventListener('resize',resizeCanvas);
resizeCanvas();

function shootConfetti(){
  const colors=['#f97316','#22c55e','#3b82f6','#eab308','#a78bfa','#ef4444'];
  for(let i=0;i<180;i++){
    confettiPieces.push({
      x:Math.random()*confettiCanvas.width,
      y:-10,
      vx:(Math.random()-0.5)*3,
      vy:Math.random()*3+2,
      size:Math.random()*6+3,
      color:colors[Math.floor(Math.random()*colors.length)],
      rot:Math.random()*Math.PI,
      vr:(Math.random()-0.5)*0.2,
      life:120+Math.random()*60
    });
  }
}
function confettiLoop(){
  ctx.clearRect(0,0,confettiCanvas.width,confettiCanvas.height);
  for(let i=confettiPieces.length-1;i>=0;i--){
    const p=confettiPieces[i];
    p.x+=p.vx;
    p.y+=p.vy;
    p.rot+=p.vr;
    p.vy+=0.03;
    p.life--;
    ctx.save();
    ctx.translate(p.x,p.y);
    ctx.rotate(p.rot);
    ctx.fillStyle=p.color;
    ctx.fillRect(-p.size/2,-p.size/2,p.size,p.size*0.6);
    ctx.restore();
    if(p.y>confettiCanvas.height+20 || p.life<=0) confettiPieces.splice(i,1);
  }
  requestAnimationFrame(confettiLoop);
}
confettiLoop();

function initStyleScoreboard(){
  styleSentWordTotal = [];
  styleSentWordChanged = [];
  styleSentChips = [];
  styleLineChips.innerHTML="";
  for(let i=0;i<STYLE_SENTENCES.length;i++){
    styleSentWordTotal.push(0);
    styleSentWordChanged.push(0);
    const chip=document.createElement('div');
    chip.className='linechip pending';
    chip.textContent=(i+1);
    styleLineChips.appendChild(chip);
    styleSentChips.push(chip);
  }
  updateStyleSentenceChips();
}

function updateStyleSentenceChips(){
  const total=STYLE_SENTENCES.length;
  let done=0;
  let partialCount=0;
  for(let i=0;i<total;i++){
    const chip=styleSentChips[i];
    const tot=styleSentWordTotal[i];
    const chg=styleSentWordChanged[i];
    chip.classList.remove('ok','partial','pending');
    if(tot===0 || chg===0){
      chip.classList.add('pending');
    }else if(chg<tot){
      chip.classList.add('partial');
      partialCount++;
    }else{
      chip.classList.add('ok');
      done++;
    }
  }
  const percent = Math.round(done/total*100);
  styleProgressText.textContent = `Zinnen af: ${done}/${total} (${percent}%)`;
  levelStats.style.percent = percent;
  levelStats.style.done = (done===total && total>0);
  if(levelStats.style.done) unlockSpeed();
}

function markWordChanged(span){
  const sIndex=parseInt(span.dataset.sent||"-1",10);
  if(isNaN(sIndex) || sIndex<0) return;
  if(span.dataset.special!=="1") return; 
  if(span.dataset.changed!=="1"){
    span.dataset.changed="1";
    styleSentWordChanged[sIndex]++;
    updateStyleSentenceChips();
  }
}

function initStyleArea(){
  styleArea.innerHTML="";
  initStyleScoreboard();

  STYLE_SENTENCES.forEach((sentence, sIndex)=>{
    const specials = (STYLE_SPECIAL_WORDS[sIndex] || []).map(w=>w.toLowerCase());
    const tokens=sentence.split(/(\s+)/);
    tokens.forEach(tok=>{
      if(tok.trim()===""){
        styleArea.appendChild(document.createTextNode(tok));
      }else{
        const span=document.createElement('span');
        span.textContent=tok;
        span.style.padding='2px 3px';
        span.style.borderRadius='6px';
        span.dataset.sent=String(sIndex);
        span.dataset.changed="0";

        const norm = tok.toLowerCase().replace(/[.,!?]/g,"");
        if(specials.includes(norm)){
          span.dataset.special="1";
          styleSentWordTotal[sIndex]++;
        }else{
          span.dataset.special="0";
        }

        span.addEventListener('click',()=> selectWord(span));
        styleArea.appendChild(span);
      }
    });
    styleArea.appendChild(document.createElement('br'));
  });
  updateStyleSentenceChips();
}

// -------- Eindresultaat (alle levels) --------
function formatTime(secs){
  secs=Math.round(secs);
  if(secs<60) return `${secs}s`;
  const m=Math.floor(secs/60);
  const s=secs%60;
  return `${m}m ${s}s`;
}
function showEndResult(){
  const totalStyle=STYLE_SENTENCES.length;
  let doneStyle=0;
  for(let i=0;i<totalStyle;i++){
    if(styleSentWordTotal[i]>0 && styleSentWordChanged[i]===styleSentWordTotal[i]) doneStyle++;
  }
  const stylePct = totalStyle>0 ? Math.round(doneStyle/totalStyle*100) : 0;
  levelStats.style.percent = stylePct;
  levelStats.style.done = (doneStyle===totalStyle && totalStyle>0);

  const accs=[];
  ['easy','normal','hard'].forEach(l=>{
    if(levelStats[l].done) accs.push(levelStats[l].acc);
  });
  let avgAcc = accs.length ? accs.reduce((a,b)=>a+b,0)/accs.length : 0;

  const metrics=[];
  if(avgAcc>0) metrics.push(avgAcc);
  if(stylePct>0) metrics.push(stylePct);
  const overallPct = metrics.length ? metrics.reduce((a,b)=>a+b,0)/metrics.length : 0;
  let grade = overallPct>0 ? overallPct/10 : 0;
  if(overallPct>0){
    grade = Math.round(grade*10)/10;
    if(grade<1) grade=1;
    if(grade>10) grade=10;
  }

  resultGrade.textContent = overallPct>0 ? grade.toFixed(1).replace('.',',') : "--";

  let label="Begin met oefenen!";
  if(overallPct>0){
    if(grade>=9) label="Topper! Fantastisch getypt!";
    else if(grade>=8) label="Heel goed gedaan!";
    else if(grade>=7) label="Mooi gewerkt!";
    else label="Goed geoefend, blijf oefenen!";
  }
  resultLabel.textContent=label;

  function levelLine(key,label){
    const st=levelStats[key];
    if(!st.done) return `<strong>${label}:</strong> nog niet afgerond.<br>`;
    return `<strong>${label}:</strong> nauwkeurigheid ${st.acc}% · tijd ${formatTime(st.time)} · fouten ${st.errors}<br>`;
  }

  let details="";
  details += levelLine('easy','Makkelijk');
  details += levelLine('normal','Normaal');
  details += levelLine('hard','Moeilijk');

  if(totalStyle===0){
    details += `<strong>Kleur typen:</strong> nog niet gestart.<br>`;
  }else{
    details += `<strong>Kleur typen:</strong> ${stylePct}% van de zinnen helemaal aangepast (groene bolletjes).<br>`;
  }

  if(overallPct===0){
    details += `<br>Tip: maak eerst Makkelijk af, probeer daarna Normaal en Moeilijk, en sluit af met Kleur typen.`;
  }else{
    details += `<br>Tip: kijk vooral naar de levels waar je nauwkeurigheid lager is, daar kun je nog extra groeien.`;
  }

  resultDetails.innerHTML=details;
  resultOverlay.style.display='flex';
  if(overallPct>0) shootConfetti();
}

function chooseLevel(level){
  // Snelheid typen heeft een eigen scherm; het vorige level blijft bewaard voor "Terug"

    SpeedGame.open();

  }
  SpeedGame.close();

    stylePhase = 'typing';
    stylePanel.style.display='none';

    inputEl.disabled=false;
    
    setTimeout(() => inputEl.focus(), 50); 
    setTarget(STYLE_TEXT);
    hintEl.innerHTML="Kleur typen (Stap 1): Typ eerst de tekst foutloos over. Pas daarna verschijnt de magische kleuren-tool!";
  }else{
    stylePanel.style.display='none';
    document.getElementById('typePanel').style.display='block';
    inputEl.disabled=false;
    setTimeout(() => inputEl.focus(), 50); 
  }

  if(level==='easy'){
    setTarget(EASY_WORDS);
    hintEl.innerHTML="Makkelijk: typ <strong>alleen woorden</strong> met <strong>spaties</strong>. Geen leestekens.";
  }else if(level==='normal'){
    setTarget(NORMAL_TEXT);
    hintEl.innerHTML="Normaal: gebruik <strong>Shift + letter = hoofdletter</strong>, <strong>punten</strong> en <strong>komma's</strong>. 10 regels — gebruik <strong>Enter</strong>.";
  }else if(level==='hard'){
    setTarget(HARD_TEXT);
    hintEl.innerHTML="Moeilijk: gebruik <strong>( )</strong>, <strong>?</strong>, <strong>/</strong> en <strong>@</strong>. 10 regels (Enter).";
  }
}
